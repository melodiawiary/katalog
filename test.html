let disk1 = []; // Utwory na płycie const STORAGE_DISK_KEY = 'melodia_wiary_disk1_storage'; // Inicjalizacja płyty z pamięci const savedDisk = localStorage.getItem(STORAGE_DISK_KEY); if(savedDisk) disk1 = JSON.parse(savedDisk); function addToDisk(tytul) { const song = cart.find(s => s.tytul === tytul); if (!song) return; // Obliczanie aktualnego czasu na płycie const currentSeconds = calculateSeconds(disk1); const songSeconds = parseDuration(song.czas || "0:00"); if (currentSeconds + songSeconds > 50 * 60) { alert("Płyta jest pełna! (Max 50 minut)"); return; } if (!disk1.some(s => s.tytul === tytul)) { disk1.push(song); localStorage.setItem(STORAGE_DISK_KEY, JSON.stringify(disk1)); render(); } } function removeFromDisk(tytul) { disk1 = disk1.filter(s => s.tytul !== tytul); localStorage.setItem(STORAGE_DISK_KEY, JSON.stringify(disk1)); render(); } function parseDuration(timeStr) { const p = timeStr.split(':'); return p.length === 2 ? parseInt(p[0]) * 60 + parseInt(p[1]) : 0; } function calculateSeconds(list) { return list.reduce((acc, s) => acc + parseDuration(s.czas || "0:00"), 0); }
